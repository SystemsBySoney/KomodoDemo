Analytic Step, SQL Query
-- Extract GLP-1 Rx volumes by NDC for T2D cohort
SELECT
  drug.brand_name,
  COUNT(DISTINCT rx.patient_token) AS patients,
  COUNT(rx.rx_id) AS total_rx
FROM PLAID.pharmacy_rx rx
JOIN PLAID.drug_dim drug ON rx.ndc = drug.ndc
JOIN PLAID.dx_history dx ON rx.patient_token = dx.patient_token
WHERE dx.dx_code LIKE 'E11%'  -- ICD-10 T2D
  AND drug.brand_name IN ('OZEMPIC', 'MOUNJARO', 'WEGOVY', ...)
  AND rx.fill_date BETWEEN '2025-05-01' AND '2026-04-30'
GROUP BY drug.brand_name;
